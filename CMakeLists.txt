cmake_minimum_required(VERSION 3.25)
project(typist101 VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

find_package(PkgConfig REQUIRED)
pkg_check_modules(FONTCONFIG REQUIRED IMPORTED_TARGET fontconfig)
pkg_check_modules(EPOXY REQUIRED IMPORTED_TARGET epoxy)
option(TYPIST_SDL_IME "Build raylib with SDL for text input and IME" ON)
option(TYPIST_PREFER_SDL3 "Use SDL3 when it is available" ON)
if(TYPIST_SDL_IME)
  if(TYPIST_PREFER_SDL3)
    find_package(SDL3 QUIET)
  endif()
  if(SDL3_FOUND)
    set(TYPIST_SDL_TARGET SDL3::SDL3)
    add_compile_definitions(TYPIST_SDL3)
  else()
    find_package(SDL2 REQUIRED)
    set(TYPIST_SDL_TARGET SDL2::SDL2)
  endif()
  set(PLATFORM SDL CACHE STRING "raylib platform" FORCE)
else()
  set(PLATFORM Desktop CACHE STRING "raylib platform" FORCE)
endif()
set(OPENGL_VERSION 4.3 CACHE STRING "raylib OpenGL version")
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CUSTOMIZE_BUILD ON CACHE BOOL "" FORCE)
set(USE_AUDIO OFF CACHE BOOL "" FORCE)
set(SUPPORT_MODULE_RAUDIO OFF CACHE BOOL "" FORCE)
set(SUPPORT_CUSTOM_FRAME_CONTROL OFF CACHE BOOL "" FORCE)
add_subdirectory(vendors/raylib)

add_library(typist_core src/session.cpp src/status.cpp src/utf8.cpp src/lesson.cpp src/text_layout.cpp src/progress.cpp)
target_include_directories(typist_core PUBLIC src)

add_executable(typist101 src/main.cpp src/visuals.cpp src/font.cpp src/slug_text.cpp src/bloom.cpp src/text_viewport.cpp src/text_input.cpp)
target_include_directories(typist101 PRIVATE vendors/slug-raylib)
if(TYPIST_SDL_IME)
  target_compile_definitions(typist101 PRIVATE TYPIST_SDL_IME)
  target_link_libraries(typist101 PRIVATE raylib ${TYPIST_SDL_TARGET})
else()
  target_link_libraries(typist101 PRIVATE raylib)
endif()
target_link_libraries(typist101 PRIVATE typist_core PkgConfig::FONTCONFIG PkgConfig::EPOXY)

include(GNUInstallDirs)
install(TARGETS typist101 RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} COMPONENT Runtime)
install(DIRECTORY lessons/ DESTINATION ${CMAKE_INSTALL_DATADIR}/typist101/lessons
        COMPONENT Runtime FILES_MATCHING PATTERN "*.txt")

include(CTest)
if(BUILD_TESTING)
  add_executable(typist_core_test tests/session_test.cpp)
  target_link_libraries(typist_core_test PRIVATE typist_core)
  add_test(NAME typist_core_test COMMAND typist_core_test)
  add_executable(typist_viewport_test tests/viewport_test.cpp src/text_viewport.cpp)
  target_link_libraries(typist_viewport_test PRIVATE typist_core)
  target_link_libraries(typist_viewport_test PRIVATE raylib PkgConfig::EPOXY)
  add_test(NAME typist_viewport_test COMMAND typist_viewport_test)
  if(TYPIST_SDL_IME)
    set_tests_properties(typist_viewport_test PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=offscreen")
  endif()
  if(TYPIST_SDL_IME)
    add_executable(typist_input_test tests/text_input_test.cpp src/text_input.cpp)
    target_include_directories(typist_input_test PRIVATE src)
    target_compile_definitions(typist_input_test PRIVATE TYPIST_SDL_IME)
    target_link_libraries(typist_input_test PRIVATE ${TYPIST_SDL_TARGET} typist_core)
    add_test(NAME typist_input_test COMMAND typist_input_test)
    add_executable(typist_bloom_test tests/bloom_test.cpp src/bloom.cpp)
    target_include_directories(typist_bloom_test PRIVATE src)
    target_link_libraries(typist_bloom_test PRIVATE raylib)
    add_test(NAME typist_bloom_test COMMAND typist_bloom_test)
    set_tests_properties(typist_bloom_test PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=offscreen")
  endif()
endif()
